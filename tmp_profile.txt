function printMemberProfile(id) {
  const m = allMembers.find(x => x.id === id);
  if (!m) return;
  const memberId = m.member_id || `GUYGD-${new Date().getFullYear()}-${String(m.id).padStart(4,'0')}`;
  const origin = window.location.origin;
  const blobUrl = toBlob(m.profile_photo);

  const photoHtml = blobUrl
    ? `<img src="${blobUrl}" style="width:90px;height:90px;border-radius:8px;object-fit:cover;border:3px solid #15803d;"/>`
    : `<div style="width:90px;height:90px;border-radius:8px;background:#2d6a4f;display:flex;align-items:center;justify-content:center;font-size:2.5rem;color:#fff;">&#128100;</div>`;

  const row = (label, val) => val ? `
    <tr>
      <td style="padding:6px 12px;color:#6b7280;font-size:0.78rem;font-weight:600;text-transform:uppercase;letter-spacing:0.4px;width:36%;border-bottom:1px solid #f3f4f6;vertical-align:top;">${label}</td>
      <td style="padding:6px 12px;color:#1a1f36;font-size:0.85rem;border-bottom:1px solid #f3f4f6;">${val}</td>
    </tr>` : '';

  const section = (title, rows) => rows.trim() ? `
    <div style="margin-bottom:16px;">
      <div style="background:#f0fdf4;border-left:4px solid #15803d;padding:6px 14px;margin-bottom:0;">
        <span style="font-size:0.72rem;font-weight:800;text-transform:uppercase;letter-spacing:1px;color:#15803d;">${title}</span>
      </div>
      <table style="width:100%;border-collapse:collapse;border:1px solid #e5e7eb;border-top:none;">${rows}</table>
    </div>` : '';

  const w = window.open('', '_blank', 'width=800,height=1000');
  w.document.write(`<!DOCTYPE html><html><head><title>GUYGD Profile - ${m.full_name}</title>
  <script src="https://cdnjs.cloudflare.com/ajax/libs/qrcodejs/1.0.0/qrcode.min.js"><\/script>
  <style>
    *{box-sizing:border-box;margin:0;padding:0;}
    body{background:#f3f4f6;font-family:'Segoe UI',Arial,sans-serif;padding:24px;}
    .page{background:#fff;max-width:780px;margin:0 auto;padding:20px 24px;border-radius:8px;box-shadow:0 2px 12px rgba(0,0,0,0.1);}
    @media print{
      body{background:#fff;padding:0;}
      .page{box-shadow:none;border-radius:0;padding:12px 16px;}
      .no-print{display:none;}
      *{-webkit-print-color-adjust:exact!important;print-color-adjust:exact!important;color-adjust:exact!important;}
    }
  </style>
  </head><body>
  <div style="max-width:780px;margin:0 auto 16px;text-align:right;" class="no-print">
    <button onclick="window.print()" style="padding:10px 28px;background:#2d6a4f;color:#fff;border:none;border-radius:8px;font-size:1rem;cursor:pointer;">&#128424; Print / Save PDF</button>
  </div>
  <div class="page">

    <!-- Header -->
    <div style="background:linear-gradient(135deg,#052e16,#15803d);border-radius:10px;padding:16px 20px;display:flex;align-items:center;gap:16px;margin-bottom:18px;">
      <div style="width:52px;height:52px;border-radius:50%;overflow:hidden;border:2px solid #d4a017;flex-shrink:0;background:#d4a017;display:flex;align-items:center;justify-content:center;">
        <img src="${origin}/assets/images/guygd-logo.png" style="width:100%;height:100%;object-fit:cover;" onerror="this.parentElement.innerHTML='<b style=color:#052e16;font-size:1.2rem>G</b>'"/>
      </div>
      <div style="flex:1;">
        <div style="color:#fff;font-size:1.1rem;font-weight:800;letter-spacing:2px;">GUYGD</div>
        <div style="color:rgba(255,255,255,0.7);font-size:0.65rem;">Gbeh-lay United Youths for Growth and Development</div>
        <div style="color:#d4a017;font-size:0.62rem;font-style:italic;margin-top:2px;">Voice of the Voiceless &mdash; Est. May 14, 2023</div>
      </div>
      <div style="text-align:right;flex-shrink:0;">
        <div style="color:rgba(255,255,255,0.55);font-size:0.6rem;text-transform:uppercase;letter-spacing:1px;">Member Profile</div>
        <div style="color:#d4a017;font-weight:800;font-family:monospace;font-size:0.85rem;margin-top:3px;">${memberId}</div>
        <div style="color:rgba(255,255,255,0.5);font-size:0.62rem;margin-top:3px;">Printed: ${new Date().toLocaleDateString()}</div>
      </div>
    </div>

    <!-- Photo + Name Banner -->
    <div style="display:flex;gap:16px;align-items:center;border:1px solid #e5e7eb;border-radius:8px;padding:14px 18px;margin-bottom:16px;background:#fafafa;">
      ${photoHtml}
      <div style="flex:1;">
        <div style="font-size:1.4rem;font-weight:800;color:#052e16;">${m.full_name}</div>
        <div style="color:#15803d;font-weight:600;font-size:0.85rem;margin-top:3px;">${m.membership_type || 'Regular Member'}</div>
        <div style="display:flex;gap:8px;margin-top:8px;flex-wrap:wrap;">
          <span style="background:${m.status==='active'?'#d4edda':'#fff3cd'};color:${m.status==='active'?'#155724':'#856404'};padding:3px 12px;border-radius:20px;font-size:0.72rem;font-weight:700;text-transform:uppercase;">${m.status}</span>
          <span style="background:#e8d5f5;color:#6f42c1;padding:3px 12px;border-radius:20px;font-size:0.72rem;font-weight:700;text-transform:capitalize;">${(m.role||'').replace('_',' ')}</span>
        </div>
      </div>
      <div id="profile-qr" style="width:80px;height:80px;flex-shrink:0;"></div>
    </div>

    <!-- Personal Info -->
    ${section('Personal Information',
      row('Full Name', m.full_name) +
      row('Gender', m.gender) +
      row('Date of Birth', m.date_of_birth ? new Date(m.date_of_birth).toLocaleDateString() : null) +
      row('Nationality', m.nationality) +
      row('Phone', m.phone) +
      row('Email', m.email)
    )}

    <!-- Location -->
    ${section('Location',
      row('Address', m.address) +
      row('Town', m.town) +
      row('District', m.district) +
      row('County', m.county)
    )}

    <!-- Background -->
    ${section('Background',
      row('Occupation', m.occupation) +
      row('Education Level', m.education_level) +
      row('Reason for Joining', m.reason_for_joining)
    )}

    <!-- Emergency Contact -->
    ${section('Emergency Contact',
      row('Name', m.emergency_contact_name) +
      row('Phone', m.emergency_contact_phone)
    )}

    <!-- Membership -->
    ${section('Membership Details',
      row('Member ID', memberId) +
      row('Membership Type', m.membership_type) +
      row('Role', (m.role||'').replace('_',' ')) +
      row('Status', m.status) +
      row('Date Joined', m.joined_at ? new Date(m.joined_at).toLocaleDateString() : '-')
    )}

    <!-- Footer -->
    <div style="margin-top:16px;padding-top:10px;border-top:1px dashed #d1fae5;display:flex;justify-content:space-between;align-items:center;">
      <div style="font-size:0.65rem;color:#9ca3af;">Confidential &mdash; issued by GUYGD Administration<br/>Gbeh-lay, Liberia &bull; info@guygd.org</div>
      <div style="font-size:0.65rem;color:#9ca3af;text-align:right;">Printed: ${new Date().toLocaleString()}<br/>Authorized use only</div>
    </div>

  </div>
  <script>
    var qrEl = document.getElementById('profile-qr');
    new QRCode(qrEl, { text: '${origin}', width: 80, height: 80, colorDark: '#052e16', colorLight: '#ffffff', correctLevel: QRCode.CorrectLevel.M });
  <\/script>
  </body></html>`);
  w.document.close();
  setTimeout(() => { if (blobUrl && blobUrl.startsWith('blob:')) URL.revokeObjectURL(blobUrl); }, 60000);
}
